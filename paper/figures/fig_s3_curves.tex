% S3 training curves for the paper (Section 6.4). Data: s3_val.dat and s3_ent.dat, written by s3_curves.py from the
% S3 run logs. Build: tectonic fig_s3_curves.tex (from this folder), then include fig_s3_curves.pdf.
\documentclass[border=2pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[dvipsnames]{xcolor}
\usepackage{pgfplots}
\usepgfplotslibrary{groupplots}
\usetikzlibrary{calc}
\pgfplotsset{compat=1.17}

\colorlet{bf}{RoyalBlue}
\colorlet{qb}{BurntOrange}
\pgfplotsset{
  bfrun/.style={bf, line width=0.6pt, mark=none},
  qbrun/.style={qb, line width=0.6pt, mark=none},
  qbstall/.style={qb, line width=0.6pt, dashed, mark=none},
}

\begin{document}
\pgfplotstableread{s3_val.dat}\valtab
\pgfplotstableread{s3_ent.dat}\enttab
\begin{tikzpicture}
\begin{groupplot}[
  group style={group size=2 by 1, horizontal sep=1.7cm},
  width=7.4cm, height=5.4cm,
  xmin=0, xmax=440, xtick={0,100,200,300,400},
  xlabel={training step},
  tick label style={font=\footnotesize}, label style={font=\footnotesize},
  title style={font=\footnotesize},
]
\nextgroupplot[ymin=0, ymax=60, ytick={0,10,20,30,40,50,60},
  ylabel={GSM8K test accuracy (\%)}, title={(a) Greedy validation accuracy},
  legend to name=s3legend, legend columns=3,
  legend style={font=\footnotesize, draw=none, /tikz/every even column/.append style={column sep=0.5cm}}]
\addlegendimage{bfrun}\addlegendentry{BF16 rollouts (R1, R2)}
\addlegendimage{qbrun}\addlegendentry{INT8 rollouts (R3--R6)}
\addlegendimage{qbstall}\addlegendentry{INT8, early stall}
\pgfplotsinvokeforeach{R3s1,R3s2,R4s1,R4s2,R4s3,R5s2,R6s1,R6s2}{\addplot[qbrun] table[x=step,y=#1]{\valtab};}
\pgfplotsinvokeforeach{R3s3,R5s1,R5s3,R6s3}{\addplot[qbstall] table[x=step,y=#1]{\valtab};}
\pgfplotsinvokeforeach{R1s1,R1s2,R1s3,R2s1,R2s2,R2s3}{\addplot[bfrun] table[x=step,y=#1]{\valtab};}

\nextgroupplot[ymin=0, ymax=0.8, ytick={0,0.2,0.4,0.6,0.8},
  yticklabel style={/pgf/number format/fixed, /pgf/number format/precision=1},
  ylabel={entropy (median over each epoch)}, title={(b) Policy entropy}]
\pgfplotsinvokeforeach{R3s1,R3s2,R4s1,R4s2,R4s3,R5s2,R6s1,R6s2}{\addplot[qbrun] table[x=step,y=#1]{\enttab};}
\pgfplotsinvokeforeach{R3s3,R5s1,R5s3,R6s3}{\addplot[qbstall] table[x=step,y=#1]{\enttab};}
\pgfplotsinvokeforeach{R1s1,R1s2,R1s3,R2s1,R2s2,R2s3}{\addplot[bfrun] table[x=step,y=#1]{\enttab};}
\end{groupplot}
\node[anchor=south] at ($(group c1r1.north)!0.5!(group c2r1.north) + (0,0.6cm)$) {\pgfplotslegendfromname{s3legend}};
\end{tikzpicture}
\end{document}
